6. Code:


#include "arduino_secrets.h"
#include "thingProperties.h"

#include <DHT.h>

#define DHTPIN 2        // GPIO2 (Change if needed)
#define DHTTYPE DHT11   // DHT11 Sensor

DHT dht(DHTPIN, DHTTYPE);

// Forward declaration (important for IoT Cloud)
void onTempChange();

void setup() {
  Serial.begin(9600);
  delay(1500);

  // IoT Cloud init
  initProperties();
  ArduinoCloud.begin(ArduinoIoTPreferredConnection);

  setDebugMessageLevel(2);
  ArduinoCloud.printDebugInfo();

  // Start DHT sensor
  Serial.println("DHT11 Sensor Test...");
  dht.begin();
}

void loop() {
  ArduinoCloud.update();

  static unsigned long lastRead = 0;

  if (millis() - lastRead >= 2000) {
    lastRead = millis();

    // Read temperature
    Temp = dht.readTemperature();

    // Check if reading failed
    if (isnan(Temp)) {
      Serial.println("Failed to read from DHT sensor!");
      return;
    }

    // Print temperature
    Serial.print("Temperature: ");
    Serial.print(Temp);
    Serial.println(" °C");
  }
}

/*
  Runs if value is changed from dashboard (optional)
*/
void onTempChange() {
  // Usually not needed for sensor (cloud → device)
}

7. Connect ESP32 to CPU, connect Sensor OUT -> D2

DHT11 Sensor  |   ESP32 Pin
-------------------------------
    VCC (+)   |      3V3
  DATA / OUT  |       D2
    GND (–)   |       GND
